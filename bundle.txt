def content_image_path(instance, filename):
    extension = Path(filename).suffix.lower()
    if isinstance(instance, CatalogItem):
        return f'catalog/{instance.category}/{uuid4().hex}{extension}'
    return f'pages/{uuid4().hex}{extension}'


class PageContent(models.Model):

        old_name = None
        if self.pk:
            old_name = PageContent.objects.filter(pk=self.pk).values_list('hero_image', flat=True).first()

        return super().delete(*args, **kwargs)


class CatalogItem(models.Model):
    page = models.CharField(max_length=40, choices=PRODUCT_PAGE_CHOICES)
    name = models.CharField(max_length=120)
    slug = models.SlugField(max_length=140, unique=True, blank=True)
    description = models.CharField(max_length=280)
    image = models.ImageField(
        upload_to=content_image_path,
        validators=[FileExtensionValidator(['jpg', 'jpeg', 'png', 'webp']), validate_image_size],
        help_text='Upload a clear photo of the actual product or branded merchandise.',
    )
    is_featured = models.BooleanField(
        default=True,
        help_text='Featured products appear on the homepage as well as their category page.',
    )
    is_published = models.BooleanField(default=True)
    display_order = models.PositiveSmallIntegerField(default=0)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['display_order', 'name']
        verbose_name = 'product / merchandise'
        verbose_name_plural = 'products & merchandise'

    def save(self, *args, **kwargs):
        old_name = None
        if self.pk:
            old_name = CatalogItem.objects.filter(pk=self.pk).values_list('image', flat=True).first()
        if not self.slug:
            base_slug = slugify(self.name)[:140] or 'product'
            candidate = base_slug
            suffix = 2
            while CatalogItem.objects.filter(slug=candidate).exclude(pk=self.pk).exists():
                ending = f'-{suffix}'
                candidate = f'{base_slug[:140 - len(ending)]}{ending}'
                suffix += 1
            self.slug = candidate
        super().save(*args, **kwargs)
        if old_name and old_name != self.image.name:
            self.image.storage.delete(old_name)

    def __str__(self):
        return self.name

    def delete(self, *args, **kwargs):
        self.image.delete(save=False)
        return super().delete(*args, **kwargs)

class PageSection(models.Model):

    title = models.CharField(max_length=140)
    body = models.TextField(max_length=1600, blank=True)
    image = models.ImageField(
        upload_to=content_image_path,
        blank=True,
        validators=[FileExtensionValidator(['jpg', 'jpeg', 'png', 'webp']), validate_image_size],
        help_text='Optional. JPG, PNG or WebP, no larger than 8 MB.',
    )
    button_text = models.CharField(max_length=80, blank=True)
    button_url = models.CharField(
        max_length=300,
        blank=True,
        help_text='Optional. Use a website path such as /contact/ or a full https:// URL.',
    )
    display_order = models.PositiveSmallIntegerField(default=0)
    is_published = models.BooleanField(default=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['page', 'display_order', 'id']
        verbose_name = 'extra page section'
        verbose_name_plural = 'extra page sections'

    def __str__(self):
        return f'{self.get_page_display()} — {self.title}'

    def clean(self):
        from django.core.exceptions import ValidationError
        from django.core.validators import URLValidator

        if self.button_url and not self.button_url.startswith('/'):
            try:
                URLValidator(schemes=['http', 'https'])(self.button_url)
            except ValidationError as exc:
                raise ValidationError({'button_url': 'Enter a website path or an http/https URL.'}) from exc
        elif self.button_url.startswith('//'):
            raise ValidationError({'button_url': 'Enter a website path beginning with one slash, not two.'})

    def save(self, *args, **kwargs):
        old_name = None
        if self.pk:
            old_name = PageSection.objects.filter(pk=self.pk).values_list('image', flat=True).first()
        super().save(*args, **kwargs)
        if old_name and old_name != self.image.name:
            self.image.storage.delete(old_name)

    def delete(self, *args, **kwargs):
        if self.image:
            self.image.delete(save=False)
        return super().delete(*args, **kwargs)


class VisualEdit(models.Model):
    FONT_CHOICES = [
        ('', 'Keep the website default'),
        ('Manrope', 'Manrope'),
        ('DM Sans', 'DM Sans'),
        ('Georgia', 'Georgia'),
        ('Arial', 'Arial'),
        ('system-ui', 'System default'),
    ]
    page = models.CharField(max_length=40, choices=VISUAL_PAGE_CHOICES)
    selector = models.CharField(max_length=1200)
    text_value = models.TextField(blank=True)
    text_enabled = models.BooleanField(default=False)
    image = models.ImageField(
        upload_to=content_image_path,
        blank=True,
        validators=[FileExtensionValidator(['jpg', 'jpeg', 'png', 'webp']), validate_image_size],
    )
    font_family = models.CharField(max_length=30, choices=FONT_CHOICES, blank=True)
    font_size = models.PositiveSmallIntegerField(
        null=True,
        blank=True,
        validators=[MinValueValidator(8), MaxValueValidator(96)],
    )
    text_color = models.CharField(
        max_length=7,
        blank=True,
        validators=[RegexValidator(r'^#[0-9a-fA-F]{6}$', 'Enter a hex colour such as #183c35.')],
    )
    hidden = models.BooleanField(default=False)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['page', 'selector']
        constraints = [
            models.UniqueConstraint(fields=['page', 'selector'], name='unique_visual_edit_per_page_element'),
        ]

    def __str__(self):
        return f'{self.get_page_display()} — {self.selector[:60]}'

    def clean(self):
        from django.core.exceptions import ValidationError

        allowed_selector = re.fullmatch(
            r'(?:main\[data-content-page="(?P<main_page>[a-z0-9-]+)"\]|header|footer|\.topline)'
            r'(?:(?: > [a-z][a-z0-9-]*:nth-of-type\([1-9]\d*\))|'
            r'(?: \[data-editor-key="[a-z0-9-]+"\]))+',
            self.selector,
        )
        if not allowed_selector:
            raise ValidationError({'selector': 'Select an element from the visual preview.'})
        main_page = allowed_selector.group('main_page')
        if (main_page and main_page != self.page) or (not main_page and self.page != 'global'):
            raise ValidationError({'page': 'This visual change belongs to a different page.'})

    def save(self, *args, **kwargs):
        old_name = None
        if self.pk:
            old_name = VisualEdit.objects.filter(pk=self.pk).values_list('image', flat=True).first()
        self.full_clean()
        super().save(*args, **kwargs)
        if old_name and old_name != self.image.name:
            self.image.storage.delete(old_name)

    def delete(self, *args, **kwargs):
        if self.image:
            self.image.delete(save=False)
        return super().delete(*args, **kwargs)

@require_GET
def public_media(request, path):
    media_root = Path(settings.MEDIA_ROOT).resolve()
    try:
        file_path = (media_root / path).resolve(strict=True)
        relative_path = file_path.relative_to(media_root)
    except (OSError, ValueError):
        raise Http404
    if not relative_path.parts or relative_path.parts[0] not in {'catalog', 'pages'} or not file_path.is_file():
        raise Http404

    content_type = mimetypes.guess_type(file_path.name)[0] or 'application/octet-stream'
    response = FileResponse(file_path.open('rb'), content_type=content_type)
    response['Cache-Control'] = 'public, max-age=3600'
    response['X-Content-Type-Options'] = 'nosniff'
    return response

@staff_member_required
@require_http_methods(['GET', 'POST', 'DELETE'])
def visual_edit_api(request, page):
    if page not in dict(PAGE_CHOICES):
        return JsonResponse({'detail': 'Choose a website page from the editor.'}, status=404)

    if request.method == 'GET':
        edits = VisualEdit.objects.filter(page__in=[page, 'global'])
        return JsonResponse({'edits': [{
            'selector': edit.selector,
            'text': edit.text_value if edit.text_enabled else '',
            'text_enabled': edit.text_enabled,
            'image': edit.image.url if edit.image else '',
            'font_family': edit.font_family,
            'font_size': edit.font_size or '',
            'text_color': edit.text_color,
            'hidden': edit.hidden,
        } for edit in edits]})

    selector = request.POST.get('selector', '') if request.method == 'POST' else request.GET.get('selector', '')
    if request.content_type == 'application/json':
        try:
            payload = json.loads(request.body)
        except json.JSONDecodeError:
            return JsonResponse({'detail': 'The editor sent invalid data.'}, status=400)
        selector = payload.get('selector', '')
    else:
        payload = request.POST

    edit_page = 'global' if selector.startswith(('header >', 'footer >', '.topline >')) else page
    if request.method == 'DELETE':
        edit = get_object_or_404(VisualEdit, page=edit_page, selector=selector)
        edit.delete()
        return JsonResponse({'ok': True})

    edit = VisualEdit.objects.filter(page=edit_page, selector=selector).first()
    if edit is None:
        edit = VisualEdit(page=edit_page, selector=selector)
    edit.text_value = payload.get('text', '')
    edit.text_enabled = str(payload.get('text_enabled', '')).lower() in {'true', '1', 'on'}
    edit.font_family = payload.get('font_family', '')
    font_size = payload.get('font_size', '')
    try:
        edit.font_size = int(font_size) if font_size else None
    except (TypeError, ValueError):
        return JsonResponse({'detail': 'Font size must be a number.'}, status=400)
    edit.text_color = payload.get('text_color', '').strip()
    edit.hidden = str(payload.get('hidden', '')).lower() in {'true', '1', 'on'}
    if 'image' in request.FILES:
        edit.image = request.FILES['image']
    elif str(payload.get('remove_image', '')).lower() in {'true', '1', 'on'}:
        edit.image = ''

    try:
        edit.save()
    except ValidationError as error:
        return JsonResponse({'detail': error.messages[0]}, status=400)

    return JsonResponse({'ok': True})

def _validate_upload(uploaded_file):
    extension = Path(uploaded_file.name.replace('\\', '/')).suffix.lower()
    if extension not in ALLOWED_UPLOAD_EXTENSIONS:
        return None, 'Upload a PDF, JPG, PNG, or WebP file.'
    if not uploaded_file.size:
        return None, 'The selected file is empty.'
    if uploaded_file.size > MAX_UPLOAD_SIZE:
        return None, 'Files must be 50 MB or smaller.'

    uploaded_file.seek(0)
    signature = uploaded_file.read(5)
    uploaded_file.seek(0)
    if extension == '.pdf':
        if not signature.startswith(b'%PDF-'):
            return None, 'The selected file is not a valid PDF.'
    else:
        from PIL import Image, UnidentifiedImageError
        try:
            image = Image.open(uploaded_file)
            image_format = image.format
            image.verify()
        except (OSError, UnidentifiedImageError, Image.DecompressionBombError):
            return None, 'The selected image could not be opened.'
        finally:
            uploaded_file.seek(0)
        expected = {'.jpg': 'JPEG', '.jpeg': 'JPEG', '.png': 'PNG', '.webp': 'WEBP'}[extension]
        if image_format != expected:
            return None, 'The file contents do not match its image extension.'
    return extension, None


def _client_ip(request):
    forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR', '')
    if forwarded_for:
        return forwarded_for.rsplit(',', 1)[-1].strip()
    return request.META.get('REMOTE_ADDR', 'unknown')


@csrf_exempt
@require_POST
def create_file_submission(request):
    ip = _client_ip(request)
    key = f'esicom-file-upload:{ip}'
    count = cache.get(key, 0)
    if count >= MAX_UPLOADS_PER_HOUR:
        return JsonResponse({'detail': 'Too many uploads from this connection. Please try again later.'}, status=429)
    if request.POST.get('website', '').strip():
        return JsonResponse({'ok': True}, status=201)
    cache.set(key, count + 1, timeout=3600)
    if request.POST.get('delivery_consent') != 'on':
        return JsonResponse({'detail': 'Confirm that Esicom may receive this file using your selected delivery method.'}, status=400)

    uploaded_file = request.FILES.get('file')
    if uploaded_file is None:
        return JsonResponse({'detail': 'Choose a file to upload.'}, status=400)
    conversion = request.POST.get('conversion', '').strip().lower()
    delivery_method = request.POST.get('delivery', 'email').strip().lower()
    if delivery_method not in {'email', 'whatsapp'}:
        return JsonResponse({'detail': 'Choose email or WhatsApp delivery.'}, status=400)
    if delivery_method == 'email' and uploaded_file.size > MAX_EMAIL_ATTACHMENT_SIZE:
        return JsonResponse({
            'detail': 'Email attachments must be 15 MB or smaller. Choose WhatsApp to share larger files.'
        }, status=413)
    extension, error = _validate_upload(uploaded_file)
    if error:
        return JsonResponse({'detail': error}, status=400)

    if conversion:
        valid_conversion = (
            conversion == 'docx' and extension == '.pdf'
        ) or (
            conversion in ALLOWED_IMAGE_CONVERSIONS and extension in {'.jpg', '.jpeg', '.png', '.webp'}
        )
        if not valid_conversion:
            return JsonResponse({'detail': 'That conversion is not available for this file type.'}, status=400)

    name = request.POST.get('name', '').strip()
    phone = request.POST.get('phone', '').strip()
    note = request.POST.get('note', '').strip()
    if len(name) > 120 or len(phone) > 40 or len(note) > 500:
        return JsonResponse({'detail': 'One of the contact fields is too long.'}, status=400)
    if phone and not re.fullmatch(r'[+()\-\d\s.]{5,40}', phone):
        return JsonResponse({'detail': 'Enter a valid phone number.'}, status=400)

    original_name = uploaded_file.name.replace('\\', '/').split('/')[-1][:255]
    submission = FileSubmission(
        original_filename=original_name,
        delivery_method=delivery_method,
        delivery_consent=True,
        name=name,
        phone=phone,
        note=note,
    )
    converted_path = None
    try:
        submission.uploaded_file.save(original_name, uploaded_file, save=False)
        submission.save()
        if conversion:
            converted_path = convert_file(submission.uploaded_file.path, original_name, conversion)
            converted_size = converted_path.stat().st_size
            if converted_size > MAX_UPLOAD_SIZE:
                converted_path.unlink(missing_ok=True)
                raise ValueError('The converted file is larger than the 50 MB limit.')
            if delivery_method == 'email' and converted_size > MAX_EMAIL_ATTACHMENT_SIZE:
                converted_path.unlink(missing_ok=True)
                raise ValueError('The converted file is too large to email as an attachment. Choose WhatsApp to share larger files.')
            with converted_path.open('rb') as converted_file:
                converted_contents = converted_file.read()
            converted_path.unlink(missing_ok=True)
            submission.converted_file.save(converted_path.name, ContentFile(converted_contents), save=False)
        submission.save()
    except ValueError as exc:
        _discard_submission(submission, converted_path)
        return JsonResponse({'detail': str(exc)}, status=422)
    except Exception:
        _discard_submission(submission, converted_path)
        logger.exception('File submission processing failed.')
        return JsonResponse({'detail': 'The file could not be processed. Please try again or send it without conversion.'}, status=500)

    download_url = request.build_absolute_uri(f'/api/file-submissions/{submission.download_token}/download/')
    if delivery_method == 'email':
        subject = f'New Esicom file submission #{submission.pk}'
        body = '\n'.join([
            f'File submission: #{submission.pk}',
            f'Original file: {original_name}',
            f'Name: {name or "Not provided"}',
            f'Phone: {phone or "Not provided"}',
            f'Note: {note or "Not provided"}',
            f'The uploaded file is attached. Private backup download link (available until staff deletes the submission): {download_url}',
            f'Staff record: {request.build_absolute_uri(f"/admin/quotes/filesubmission/{submission.pk}/change/")}',
        ])
        attachment = submission.converted_file or submission.uploaded_file
        email = EmailMessage(
            subject,
            body,
            settings.DEFAULT_FROM_EMAIL,
            [settings.ESICOM_ENQUIRY_EMAIL],
        )
        if settings.EMAIL_BACKEND == 'django.core.mail.backends.console.EmailBackend':
            logger.error('File submission %s was saved, but email delivery is not configured.', submission.pk)
            return JsonResponse({
                'detail': f'The file was saved (reference ES-F{submission.pk}), but email is not configured on this server. Select WhatsApp or configure SMTP.',
                'saved': True,
                'reference': submission.pk,
            }, status=503)
        try:
            email.attach_file(attachment.path)
            email.send(fail_silently=False)
        except Exception:
            logger.exception('File submission %s was saved but attachment email delivery failed.', submission.pk)
            return JsonResponse({
                'detail': f'The file was saved (reference ES-F{submission.pk}), but email delivery failed. Try WhatsApp sharing or try again.',
                'saved': True,
                'reference': submission.pk,
            }, status=502)

    return JsonResponse({
        'ok': True,
        'reference': submission.pk,
        'filename': submission.converted_file.name and Path(submission.converted_file.name).name or original_name,
        'converted': bool(submission.converted_file),
        'delivery': delivery_method,
        'download_url': f'/api/file-submissions/{submission.download_token}/download/',
    }, status=201)


def _discard_submission(submission, converted_path):
    if converted_path and converted_path.exists():
        converted_path.unlink()
    if submission.pk:
        submission.delete()
        return
    for file_field in (submission.uploaded_file, submission.converted_file):
        if file_field:
            file_field.delete(save=False)


@require_GET
def download_file_submission(request, token):
    submission = get_object_or_404(FileSubmission, download_token=token)
    file_field = submission.converted_file or submission.uploaded_file
    return _file_response(file_field)


def download_submission_attachment(request, pk, file_kind):
    submission = get_object_or_404(FileSubmission, pk=pk)
    if file_kind == 'original':
        file_field = submission.uploaded_file
    elif file_kind == 'converted' and submission.converted_file:
        file_field = submission.converted_file
    else:
        raise Http404
    return _file_response(file_field)


def _file_response(file_field):
    file_name = Path(file_field.name).name
    content_type = mimetypes.guess_type(file_name)[0] or 'application/octet-stream'
    response = FileResponse(file_field.open('rb'), as_attachment=True, filename=file_name, content_type=content_type)
    response['Cache-Control'] = 'private, no-store'
    response['X-Content-Type-Options'] = 'nosniff'
    return response

@require_POST
def create_quote(request):
    # Small IP-based throttle; configure a shared cache (e.g. Redis) for production multi-worker deployments.
    ip = request.META.get('REMOTE_ADDR', 'unknown')
    key = f'esicom-quote:{ip}'
    count = cache.get(key, 0)
    if count >= 8:
        return JsonResponse({'detail': 'Too many enquiries from this connection. Please email us instead.'}, status=429)
    try:
        payload = json.loads(request.body.decode('utf-8'))
    except (json.JSONDecodeError, UnicodeDecodeError):
        return JsonResponse({'detail': 'Please submit a valid enquiry.'}, status=400)
    if not isinstance(payload, dict):
        return JsonResponse({'detail': 'Please submit a valid enquiry.'}, status=400)
    if str(payload.get('website', '')).strip():
        return JsonResponse({'detail': 'Enquiry received.'}, status=200)
    name = str(payload.get('name', '')).strip()
    company = str(payload.get('company', '')).strip()
    phone = str(payload.get('phone', '')).strip()
    email = str(payload.get('email', '')).strip()
    needs = str(payload.get('needs', '')).strip()
    quantity = str(payload.get('quantity', '')).strip()
    source = str(payload.get('source', 'website')).strip()[:40] or 'website'
    if len(name) < 2 or len(name) > 120:
        return JsonResponse({'detail': 'Enter your name (at least 2 characters).'}, status=400)
    if not phone or len(phone) > 40 or not re.fullmatch(r'[+()\-\d\s.]{5,40}', phone):
        return JsonResponse({'detail': 'Enter a valid phone number.'}, status=400)
    if email and (len(email) > 254 or not re.fullmatch(r'[^\s@]+@[^\s@]+\.[^\s@]+', email)):
        return JsonResponse({'detail': 'Enter a valid email address.'}, status=400)
    if len(needs) < 8 or len(needs) > 5000:
        return JsonResponse({'detail': 'Please describe what you need (8–5000 characters).'}, status=400)
    if len(company) > 160 or len(quantity) > 100:
        return JsonResponse({'detail': 'One of the fields is too long.'}, status=400)
    required_date = None
    raw_date = str(payload.get('date', '')).strip()
    if raw_date:
        try:
            required_date = date.fromisoformat(raw_date)
        except ValueError:
            return JsonResponse({'detail': 'Enter a valid required date.'}, status=400)
    quote = QuoteRequest.objects.create(name=name, company=company, phone=phone, email=email, needs=needs, quantity=quantity, required_date=required_date, source=source)
    cache.set(key, count + 1, timeout=3600)
    subject = f'New Esicom website quotation request #{quote.pk}'
    body = '\n'.join([
        f'Enquiry reference: #{quote.pk}', f'Name: {name}', f'Company: {company or "Not provided"}',
        f'Phone: {phone}', f'Email: {email or "Not provided"}', f'Quantity: {quantity or "Not provided"}',
        f'Required date: {required_date or "Not provided"}', '', 'Request details:', needs,
    ])
    try:
        send_mail(subject, body, settings.DEFAULT_FROM_EMAIL, [settings.ESICOM_ENQUIRY_EMAIL], fail_silently=False)
    except Exception:
        # Persist the enquiry even if email credentials or the mail server are not configured.
        logger.exception('Quote request %s was saved but notification email failed.', quote.pk)
    return JsonResponse({'ok': True, 'reference': quote.pk}, status=201)

      const empty=document.querySelector('#empty-selection');
      const status=document.querySelector('#status');
      const imagePreview=document.querySelector('#image-preview');
      const imageFile=document.querySelector('#image-file');
      let selected=null;
      let savedEdit=null;
      let editsBySelector=new Map();
      const previewBase="{{ preview_url|escapejs }}";
      const editorOrigin="{{ editor_origin|escapejs }}";
      const apiForPage=page=>'/api/visual-edits/'+encodeURIComponent(page)+'/';
      const csrfToken=document.querySelector('meta[name="csrf-token"]').content;
      const pageUrl=page=>{
        const path=page==='home'?'/':'/'+page+'/';
        return previewBase+path+'?editor=1&editor_origin='+encodeURIComponent(new URL(editorOrigin).origin);
      };
      const setStatus=(message,isError=false)=>{
        status.textContent=message;
        status.classList.toggle('error',isError);
      };
      const refreshFrame=()=>{frame.src=pageUrl(pagePicker.value)};
      const loadPageEdits=async()=>{
        const response=await fetch(apiForPage(pagePicker.value),{headers:{'Accept':'application/json'}});
        if(!response.ok)throw new Error('Could not load saved changes for this page.');
        const result=await response.json();
        editsBySelector=new Map(result.edits.map(edit=>[edit.selector,edit]));
      };
      const previewChange=()=>{
        if(!selected)return;
        const image=imageFile.files[0];
        const sendPreview=previewImage=>{
        const edit={
          selector:selected.selector,
          text:document.querySelector('#text-enabled').checked?document.querySelector('#text-value').value:null,
          image:previewImage??(document.querySelector('#remove-image').checked?'':null),
          font_family:document.querySelector('#font-family').value,
          font_size:document.querySelector('#font-size').value,
          text_color:document.querySelector('#text-color').value,
          hidden:document.querySelector('#hidden').checked,
        };
        frame.contentWindow.postMessage({type:'esicom-visual-editor-preview',edit},new URL(previewBase).origin);
        };
        if(image){
          const reader=new FileReader();
          reader.addEventListener('load',()=>sendPreview(String(reader.result)));
          reader.addEventListener('error',()=>setStatus('The selected photo could not be previewed.',true));
          reader.readAsDataURL(image);
          return;
        }
        sendPreview(null);
      };
      const showSelection=async info=>{
        selected=info;
        if(!editsBySelector.has(info.selector))await loadPageEdits();
        savedEdit=editsBySelector.get(info.selector)||null;
        document.querySelector('#selection-name').textContent=(info.tag==='img'?'Photo':'Text or element')+' selected';
        document.querySelector('#selection-path').textContent=info.tag.toUpperCase()+' on '+pagePicker.options[pagePicker.selectedIndex].text;
        empty.hidden=true;
        form.hidden=false;
        const textEnabled=document.querySelector('#text-enabled');
        textEnabled.checked=savedEdit?savedEdit.text_enabled:info.tag!=='img';
        document.querySelector('#text-value').value=savedEdit?.text_enabled?savedEdit.text:info.text;
        document.querySelector('#text-control').hidden=info.tag==='img';
        document.querySelector('#image-control').hidden=info.tag!=='img';
        document.querySelector('#font-family').value=savedEdit?.font_family||'';
        document.querySelector('#font-size').value=savedEdit?.font_size||'';
        document.querySelector('#text-color').value=savedEdit?.text_color||'';
        document.querySelector('#hidden').checked=Boolean(savedEdit?.hidden);
        document.querySelector('#remove-image').checked=false;
        imageFile.value='';
        const image=savedEdit?.image||info.image;
        imagePreview.src=image;
        imagePreview.style.display=info.tag==='img'&&image?'block':'none';
        setStatus('');
      };
      pagePicker.addEventListener('change',async()=>{
        selected=null;
        form.hidden=true;
        sectionForm.hidden=true;
        empty.hidden=false;
        setStatus('');
        refreshFrame();
        try{await loadPageEdits()}catch(error){setStatus(error.message,true)}
      });
      window.addEventListener('message',event=>{
        if(event.source!==frame.contentWindow||event.origin!==new URL(previewBase).origin)return;
        if(event.data?.type==='esicom-visual-editor-select'&&event.data.page===pagePicker.value){
          showSelection(event.data).catch(error=>setStatus(error.message,true));
        }
      });
      form.addEventListener('input',previewChange);
      form.addEventListener('change',previewChange);
      document.querySelector('#color-picker').addEventListener('input',event=>{

      });

          imagePreview.src=URL.createObjectURL(file);
          imagePreview.style.display='block';
        }
        previewChange();
      });

        event.preventDefault();
        if(!selected)return;
        const data=new FormData();
        data.set('selector',selected.selector);
        data.set('text',document.querySelector('#text-value').value);
        data.set('text_enabled',document.querySelector('#text-enabled').checked?'true':'false');
        data.set('font_family',document.querySelector('#font-family').value);
        data.set('font_size',document.querySelector('#font-size').value);
        data.set('text_color',document.querySelector('#text-color').value);
        data.set('hidden',document.querySelector('#hidden').checked?'true':'false');
        if(imageFile.files[0])data.set('image',imageFile.files[0]);
        if(document.querySelector('#remove-image').checked)data.set('remove_image','true');
        const button=document.querySelector('#save-button');
        button.disabled=true;
        setStatus('Saving your change…');
        try{
          const response=await fetch(apiForPage(pagePicker.value),{
            method:'POST',
            headers:{'X-CSRFToken':csrfToken,'Accept':'application/json'},
            body:data,
          });
          const result=await response.json();
          if(!response.ok)throw new Error(result.detail||'The change could not be saved.');
          setStatus('Saved. This change is now part of the website.');
          await loadPageEdits();
          imageFile.value='';
          document.querySelector('#remove-image').checked=false;
          refreshFrame();
        }catch(error){setStatus(error.message,true)}
        finally{button.disabled=false}
      });
      addSectionButton.addEventListener('click',()=>{
        selected=null;
        form.hidden=true;
        empty.hidden=true;
        sectionForm.hidden=false;
        setStatus('Add a new text or photo section to this page.');
        document.querySelector('#section-title').focus();
      });
      sectionForm.addEventListener('submit',async event=>{
        event.preventDefault();

        button.disabled=true;
        setStatus('Adding your section…');

          const response=await fetch('/api/page-sections/'+encodeURIComponent(pagePicker.value)+'/',{
            method:'POST',
            headers:{'X-CSRFToken':csrfToken,'Accept':'application/json'},
            body:new FormData(sectionForm),
          });
          const result=await response.json();
          if(!response.ok)throw new Error(result.detail||'The section could not be added.');
          sectionForm.reset();
          setStatus('Section added to the page.');
          refreshFrame();
        }catch(error){setStatus(error.message,true)}
        finally{button.disabled=false}
      });
      document.querySelector('#reset-button').addEventListener('click',async()=>{
        if(!selected)return;
        if(!window.confirm('Restore this item to its original website content?'))return;
        try{
          const response=await fetch(apiForPage(pagePicker.value)+'?selector='+encodeURIComponent(selected.selector),{
            method:'DELETE',
            headers:{'X-CSRFToken':csrfToken,'Accept':'application/json'},
          });
          const result=await response.json();
          if(!response.ok)throw new Error(result.detail||'The original could not be restored.');
          setStatus('Original website content restored.');
          await loadPageEdits();
          refreshFrame();
          showSelection(selected);
        }catch(error){setStatus(error.message,true)}
      });
      refreshFrame();
      loadPageEdits().catch(error=>setStatus(error.message,true));

  if(/^(?:https?:|blob:|data:)/i.test(path))return path;
  const base=apiBase&&path.startsWith('/media/')?apiBase:location.origin;
  return new URL(path,base).href;
};

const visualOriginals=new Map();
const editorSelector=element=>{
  const root=element.closest('main[data-content-page],header,footer,.topline');
  if(!root)return '';
  const rootSelector=root.matches('main[data-content-page]')
    ?'main[data-content-page="'+pageKey+'"]'
    :root.matches('.topline')?'.topline':root.localName;
  const marker=element.closest('[data-editor-key]');
  const stableMarker=marker&&root.contains(marker)?marker:null;
  const parts=[];
  let current=element;
  const stopAt=stableMarker||root;
  while(current&&current!==stopAt){
    const siblings=[...current.parentElement.children].filter(sibling=>sibling.tagName===current.tagName);
    parts.unshift(current.localName+':nth-of-type('+(siblings.indexOf(current)+1)+')');
    current=current.parentElement;
  }
  if(stableMarker){
    const markerSelector='[data-editor-key="'+stableMarker.dataset.editorKey+'"]';
    return rootSelector+' '+markerSelector+(parts.length?' > '+parts.join(' > '):'');
  }
  return current===root?rootSelector+' > '+parts.join(' > '):'';
};
const directText=element=>[...element.childNodes]
  .filter(node=>node.nodeType===Node.TEXT_NODE)
  .map(node=>node.textContent)
  .join('')
  .trim();
const rememberOriginal=target=>{
  const selector=editorSelector(target);
  if(selector&&!visualOriginals.has(selector)){
    visualOriginals.set(selector,{
      textNodes:[...target.childNodes].filter(node=>node.nodeType===Node.TEXT_NODE).map(node=>node.textContent),
      image:target instanceof HTMLImageElement?target.getAttribute('src'):'',
      fontFamily:target.style.fontFamily,
      fontSize:target.style.fontSize,
      color:target.style.color,
      display:target.style.display,
    });
  }
  return visualOriginals.get(selector);
};
const setDirectText=(target,value)=>{
  const textNodes=[...target.childNodes].filter(node=>node.nodeType===Node.TEXT_NODE);
  if(textNodes.length){
    textNodes[0].textContent=value;
    textNodes.slice(1).forEach(node=>node.remove());
  }else target.insertBefore(document.createTextNode(value),target.firstChild);
};
const applyVisualEdits=edits=>{
  for(const edit of edits){
    if(edit.page!==pageKey&&edit.page!=='global')continue;
    const target=document.querySelector(edit.selector);
    if(!target)continue;
    rememberOriginal(target);
    if(edit.text!==null)setDirectText(target,edit.text);
    if(edit.image&&target instanceof HTMLImageElement)target.src=imageUrl(edit.image);
    if(edit.font_family)target.style.fontFamily='"'+edit.font_family+'"';
    if(edit.font_size)target.style.fontSize=edit.font_size+'px';
    if(edit.text_color)target.style.color=edit.text_color;
    if(edit.hidden)target.style.display='none';
  }
};
if(visualEditorMode){
  document.body.classList.add('visual-editor-mode');
  window.addEventListener('message',event=>{
    if(event.origin!==editorParentOrigin||event.source!==parent||event.data?.type!=='esicom-visual-editor-preview')return;
    const edit=event.data.edit;
    const target=document.querySelector(edit.selector);
    if(!target)return;
    const original=rememberOriginal(target);
    if(edit.text===null){
      const textNodes=[...target.childNodes].filter(node=>node.nodeType===Node.TEXT_NODE);
      textNodes.forEach(node=>node.remove());
      original.textNodes.forEach(text=>target.insertBefore(document.createTextNode(text),target.firstChild));
    }else setDirectText(target,edit.text);
    if(target instanceof HTMLImageElement){
      if(edit.image!==null)target.src=edit.image?imageUrl(edit.image):imageUrl(original.image);
    }
    target.style.fontFamily=edit.font_family?'"'+edit.font_family+'"':original.fontFamily;
    target.style.fontSize=edit.font_size?edit.font_size+'px':original.fontSize;
    target.style.color=edit.text_color||original.color;
    target.style.display=edit.hidden?'none':original.display;
  });
  document.addEventListener('click',event=>{
    const target=event.target.closest('img,h1,h2,h3,p,a,button,span,small,strong,b,li,.eyebrow,.mockup-label');
    if(!target||!target.closest('main[data-content-page],header,footer,.topline'))return;
    event.preventDefault();
    event.stopImmediatePropagation();
    document.querySelector('.visual-editor-selected')?.classList.remove('visual-editor-selected');
    target.classList.add('visual-editor-selected');
    if(!editorParentOrigin)return;
    parent.postMessage({
      type:'esicom-visual-editor-select',
      page:pageKey,
      selector:editorSelector(target),
      tag:target.tagName.toLowerCase(),
      text:directText(target),
      image:target instanceof HTMLImageElement?target.src:'',
      alt:target instanceof HTMLImageElement?target.alt:'',
    },editorParentOrigin);
  },true);
}

const loadSiteContent=async()=>{
  if(!main)return;
  const response=await fetch(apiBase+'/api/site-content/',{headers:{'Accept':'application/json'},cache:'no-store'});
  if(!response.ok)throw new Error('Site content service returned HTTP '+response.status);
  const content=await response.json();
  photoOverrides=content.photos||{};
  document.querySelectorAll('img[src]').forEach(image=>{
    const source=image.getAttribute('src');
    if(source&&photoOverrides[source])image.src=imageUrl(photoOverrides[source]);
  });
  const page=content.pages?.[pageKey];
  if(page){
    const eyebrow=main.querySelector('[data-copy-eyebrow]');
    if(eyebrow)eyebrow.textContent=page.eyebrow;
    const intro=main.querySelector('[data-copy-intro]');
    if(intro)intro.textContent=page.intro;
    const heading=main.querySelector('[data-copy-heading]');
    if(heading){
      const highlight=heading.querySelector('[data-copy-highlight]');
      if(highlight){
        const leadingText=[...heading.childNodes].find(node=>node.nodeType===Node.TEXT_NODE);
        if(leadingText)leadingText.textContent=page.title;
        highlight.textContent=page.highlight||'';
        highlight.hidden=!page.highlight;
      }else heading.textContent=page.title;
    }
    if(page.image){
      if(pageKey==='home')homePageImage=page.image;
      else{
        const heroImage=main.querySelector('[data-copy-image]');
        if(heroImage)heroImage.src=imageUrl(page.image);
      }
    }
  }
  const products=Array.isArray(content.products)?content.products:[];
  if(pageKey==='home'){
    const featured=products.filter(product=>product.featured);
    const container=main.querySelector('[data-featured-products]');
    if(container)container.replaceChildren(...featured.map(productCard));
    startHeroCarousel(featured);
  }else{
    const categoryProducts=products.filter(product=>product.page===pageKey);
    const section=main.querySelector('[data-category-products]');
    const container=section?.querySelector('[data-product-list]');
    if(container&&categoryProducts.length){
      container.replaceChildren(...categoryProducts.map(productCard));
      section.hidden=false;
    }
  }
  const sections=(Array.isArray(content.sections)?content.sections:[])
    .filter(section=>section.page===pageKey)
    .map(editableSection);
  main.append(...sections);
  applyVisualEdits(content.visual_edits||[]);
};

    # Build gate: all expected pages, key facts, internal targets and assets must exist.
    required=['index.html','office-supplies/index.html','computer-consumables/index.html','bulk-printing/index.html','branding/index.html','about/index.html','contact/index.html','send-file/index.html','file-qr/index.html','catalogue/esicom-suppliers-catalogue.pdf','images/photography/workspace.webp','images/photography/stationery.webp','images/photography/computer.webp','images/photography/printing.webp','images/photography/apparel.webp','images/photography/meeting-room.webp','images/photography/notebook.webp','images/photography/stock-boxed-pens.webp','images/photography/stock-copy-paper.webp','styles.css','site.js']
    missing=[x for x in required if not (DIST/x).is_file()]
    if missing: raise SystemExit('BUILD FAILED: missing files: '+', '.join(missing))
    bad=['10+ years','500+ clients','same-day delivery','nationwide delivery','SME support','authorized distributor','certifications']
    content='\n'.join(p.read_text(errors='ignore') for p in DIST.rglob('*.html')).lower()
    found=[x for x in bad if x in content]
    if found: raise SystemExit('BUILD FAILED: unsupported claims found: '+', '.join(found))
    if '/images/mockups/' in content or 'images.unsplash.com/' in content:
        raise SystemExit('BUILD FAILED: pages must use local photography assets, not mockups or remote photos')
    for path in PAGES:
        p = DIST/'index.html' if path=='/' else DIST/path.strip('/')/'index.html'
        txt=p.read_text()
        if PAGES[path]['kind']=='upload':
            if any(value not in txt for value in ('data-file-upload-form', '15 MB', '50 MB', 'delivery_consent', 'until Esicom staff deletes them')):
                raise SystemExit('BUILD FAILED: file upload page is missing its delivery options, consent, or stated limits.')
            continue
        if PAGES[path]['kind']=='qr':
            if 'data-upload-qr' not in txt or 'data-print-qr' not in txt:
                raise SystemExit('BUILD FAILED: printable file-upload QR page is incomplete.')
            continue
        if 'esicomsuppliers@gmail.com' not in txt or '0728 942 669' not in txt:
            raise SystemExit(f'BUILD FAILED: contact detail missing from {p.relative_to(DIST)}')
        page_key='home' if path=='/' else path.strip('/')
        if f'data-content-page="{page_key}"' not in txt or 'data-copy-image' not in txt:
            raise SystemExit(f'BUILD FAILED: CMS page or hero-image wiring missing from {p.relative_to(DIST)}')
        if PAGES[path]['kind']=='home' and 'data-featured-products' not in txt:
            raise SystemExit('BUILD FAILED: homepage product listing is not CMS-ready')
        if PAGES[path]['kind']=='detail' and 'data-category-products' not in txt:
            raise SystemExit(f'BUILD FAILED: category product listing is not CMS-ready on {p.relative_to(DIST)}')
    print(f'BUILD OK: {len(list(DIST.rglob("index.html")))} HTML pages, catalogue PDF and locally stored photography written to {DIST}')

if __name__=='__main__': main()
